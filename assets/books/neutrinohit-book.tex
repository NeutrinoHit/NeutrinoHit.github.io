% Short remarks rendered in the outer margin of the printed book. In HTML
% and EPUB the Lua filter turns the same source block into a disclosure.
\newcommand{\bookmarginnote}[2][]{%
  \marginpar{%
    \begin{tcolorbox}[
      enhanced,
      width=\marginparwidth,
      colback=BookSticky,
      colframe=BookStickyBorder!28!BookSticky,
      interior style={top color=BookSticky,bottom color=BookStickyDeep},
      boxrule=0.25pt,
      arc=0.6mm,
      left=2mm,
      right=2mm,
      top=2.6mm,
      bottom=2mm,
      boxsep=0pt,
      drop fuzzy shadow=BookInk!35!white,
      overlay={%
        \fill[BookStickyTape,opacity=0.9,rounded corners=0.7mm]
          ([xshift=-5.5mm,yshift=-0.8mm]frame.north)
          rectangle
          ([xshift=5.5mm,yshift=0.8mm]frame.north);%
      }
    ]%
      \raggedright\setlength{\parindent}{0pt}%
      \sffamily\footnotesize\color{BookStickyInk}%
      \if\relax\detokenize{#1}\relax\else
        {\bfseries\color{BookStickyInk}#1\par}\vspace{0.5mm}%
      \fi
      #2%
    \end{tcolorbox}%
  }%
}

% Numbered chapter opener with a local table of contents.
\renewcommand*{\chapterformat}{}
\renewcommand*{\chapterlinesformat}[3]{%
  \ifnum\value{chapter}>0\relax
    \parbox[t]{\linewidth}{%
      {\sffamily\bfseries\color{\bookchaptercolor}%
       \fontsize{15}{18}\selectfont \BookChapterLabel~\thechapter\par}%
      \vspace{7mm}%
      {\RaggedRight\sffamily\bfseries\color{BookInk}%
       \hyphenpenalty=10000\exhyphenpenalty=10000\pretolerance=10000%
       \let\BookTitleBreak\\%
       \fontsize{32}{36}\selectfont #3\par}%
      \vspace{9mm}%
      {\noindent\color{\bookchaptercolor}\rule{\linewidth}{1.3mm}\par}%
    }%
  \else
    \parbox[t]{\linewidth}{#3}%
  \fi
}

% A quiet outer-edge navigator. The thin strip follows the current topic. The
% folio is deliberately unframed: a large, light numeral in the outer margin.
\newif\ifbookpagenavigator
\bookpagenavigatortrue
\newlength{\bookpageedgewidth}
\newlength{\bookfoliooffset}
\newlength{\bookfoliotopoffset}
\setlength{\bookpageedgewidth}{1.5mm}
\setlength{\bookfoliooffset}{20mm}
\setlength{\bookfoliotopoffset}{15mm}

\newcommand{\BookFolioNumber}[1]{%
  \node[inner sep=0pt,font=\sffamily\fontsize{16}{16}\selectfont,
        text=BookMuted] at #1 {\thepage};
}

\newcommand{\BookPageNavigatorLeft}{%
  \fill[\bookchaptercolor]
    (current page.north west) rectangle
    ([xshift=\bookpageedgewidth]current page.south west);
  \BookFolioNumber{([xshift=\bookfoliooffset,yshift=-\bookfoliotopoffset]current page.north west)}%
}

\newcommand{\BookPageNavigatorRight}{%
  \fill[\bookchaptercolor]
    ([xshift=-\bookpageedgewidth]current page.north east) rectangle
    (current page.south east);
  \BookFolioNumber{([xshift=-\bookfoliooffset,yshift=-\bookfoliotopoffset]current page.north east)}%
}

\newcommand{\BookRenderPageNavigator}{%
  \ifbookpagenavigator
    \ifnum\value{part}>0\relax
      \begin{tikzpicture}[remember picture,overlay]
        \ifodd\value{page}
          \BookPageNavigatorRight
        \else
          \BookPageNavigatorLeft
        \fi
      \end{tikzpicture}%
    \fi
  \fi
}
\AddToHook{shipout/foreground}{\BookRenderPageNavigator}
